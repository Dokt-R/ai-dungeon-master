    character_knowledge: Dict[str, List[str]] = Field(..., description="Character-specific knowledge")

    summary: str = Field(..., description="Condensed memory summary for AI")
    token_count: int = Field(..., description="Estimated token count for context")

class MemoryConfig(BaseModel):
    """Configuration for memory management."""
    max_context_tokens: int = Field(default=2000, description="Maximum tokens for AI context")
    memory_relevance_threshold: float = Field(default=0.7, description="Minimum relevance score")
    max_memory_items: int = Field(default=50, description="Maximum memory items to retain")
    cleanup_interval: int = Field(default=3600, description="Cleanup interval in seconds")
