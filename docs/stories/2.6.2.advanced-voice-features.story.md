### Data Models
**Advanced Voice Models:**
```python
class SpeakerProfile(BaseModel):
    """Voice profile for speaker identification."""
    profile_id: str = Field(..., description="Unique speaker profile identifier")
    user_id: str = Field(..., description="Associated user ID")
    voice_print: bytes = Field(..., description="Voice biometric data")
    voice_characteristics: Dict[str, float] = Field(default_factory=dict)
    sample_audio_clips: List[bytes] = Field(default_factory=list)
    confidence_threshold: float = Field(default=0.8, description="Identification confidence")
    created_at: datetime = Field(default_factory=datetime.utcnow)
    last_updated: datetime = Field(default_factory=datetime.utcnow)

class VoiceActivitySegment(BaseModel):
    """Voice activity detection segment."""
    segment_id: str = Field(..., description="Unique segment identifier")
    session_id: str = Field(..., description="Voice session identifier")
    speaker_id: Optional[str] = Field(None, description="Identified speaker")
    start_time: datetime = Field(..., description="Segment start time")
    end_time: datetime = Field(..., description="Segment end time")
    confidence: float = Field(..., ge=0.0, le=1.0, description="VAD confidence")
    energy_level: float = Field(..., description="Audio energy level")
    noise_level: float = Field(..., description="Background noise level")
    overlap_detected: bool = Field(default=False, description="Speaker overlap detected")

class MultiUserConversation(BaseModel):
    """Multi-user conversation context."""
    conversation_id: str = Field(..., description="Unique conversation identifier")
    session_id: str = Field(..., description="Voice session identifier")
    active_speakers: List[str] = Field(default_factory=list, description="Currently active speakers")
    speaker_segments: List[VoiceActivitySegment] = Field(default_factory=list)
    conversation_flow: List[Dict[str, Any]] = Field(default_factory=list)
    turn_taking_events: List[Dict[str, Any]] = Field(default_factory=list)
    engagement_metrics: Dict[str, float] = Field(default_factory=dict)
    started_at: datetime = Field(default_factory=datetime.utcnow)

class AudioMixConfiguration(BaseModel):
    """Audio mixing configuration for multi-user scenarios."""
    mix_id: str = Field(..., description="Unique mix configuration identifier")
    session_id: str = Field(..., description="Voice session identifier")
    input_streams: List[str] = Field(..., description="Input audio stream IDs")
    output_stream: str = Field(..., description="Output mixed stream ID")
    volume_levels: Dict[str, float] = Field(default_factory=dict, description="Per-stream volume")
    spatial_positions: Dict[str, Dict[str, float]] = Field(default_factory=dict)
    priority_speakers: List[str] = Field(default_factory=list, description="Priority speaker order")
    ducking_enabled: bool = Field(default=True, description="Enable audio ducking")
    ducking_threshold: float = Field(default=0.7, description="Ducking trigger threshold")
```
